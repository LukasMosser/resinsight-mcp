const fs = require('node:fs');
const path = require('node:path');
const { spawn, execFileSync } = require('node:child_process');
const { chromium } = require('playwright');

const repository = '/private/tmp/resinsight-mission-docs';
const output = '/private/tmp/resinsight-mission-layout-review';
fs.mkdirSync(output, { recursive: true });
const server = spawn(path.join(repository, '.venv/bin/python'), [
  '-u', '-m', 'http.server', '0', '--bind', '127.0.0.1', '--directory', path.join(repository, 'site'),
], { stdio: ['ignore', 'pipe', 'pipe'] });
let browser;
let serverLog = '';
server.stderr.on('data', chunk => { serverLog += chunk.toString(); });

(async () => {
  try {
    const port = await new Promise((resolve, reject) => {
      const timer = setTimeout(() => reject(new Error('Local documentation server did not start.')), 10000);
      server.once('error', reject);
      server.stdout.on('data', chunk => {
        const text = chunk.toString();
        serverLog += text;
        const match = /port (\d+)/.exec(text);
        if (match) { clearTimeout(timer); resolve(Number(match[1])); }
      });
    });
    const origin = `http://127.0.0.1:${port}`;
    browser = await chromium.launch({
      executablePath: '/Applications/Google Chrome.app/Contents/MacOS/Google Chrome',
      chromiumSandbox: true,
      headless: true,
    });
    const context = await browser.newContext({ viewport: { width: 1440, height: 1050 }, reducedMotion: 'reduce' });
    await context.route('**/*', route => new URL(route.request().url()).origin === origin ? route.continue() : route.abort());
    const page = await context.newPage();
    const errors = [];
    const localLinks = new Set();
    page.on('pageerror', error => errors.push(String(error)));
    const pages = [];
    for (const item of [
      { path: '/mcp/', image: 'connection.png' },
      { path: '/tutorials/sessions/', heading: 'Open, save, and reopen', image: 'session-tutorial.png' },
      { path: '/tutorials/views/', text: 'These captures compare initial and final pressure with the same legend bounds.', image: 'view-comparison.png' },
      { path: '/development/agent-workflows/', heading: 'Gaps and proposed ownership', image: 'integration-ownership.png' },
      { path: '/development/jobs/', heading: 'Transport and review', image: 'job-transport.png' },
    ]) {
      const response = await page.goto(`${origin}${item.path}`, { waitUntil: 'networkidle' });
      if (response.status() !== 200) throw new Error(`Unexpected page status: ${response.status()} for ${item.path}`);
      await page.evaluate(() => document.fonts.ready);
      if (item.heading || item.text) {
        const target = item.heading ? page.getByRole('heading', { name: item.heading, exact: true }) : page.getByText(item.text, { exact: false });
        await target.evaluate(element => element.scrollIntoView({ block: 'start' }));
        await page.evaluate(() => window.scrollBy(0, -96));
      }
      await page.screenshot({ path: path.join(output, item.image) });
      const layout = await page.evaluate(() => ({
        viewport: innerWidth,
        document: document.documentElement.scrollWidth,
        scroll_regions: [...document.querySelectorAll('article table, article pre, article pre code')].filter(item => item.scrollWidth > item.clientWidth + 1).map(item => ({ tag: item.tagName, width: item.clientWidth, scroll_width: item.scrollWidth })),
      }));
      if (layout.document > layout.viewport) throw new Error(`The page overflows horizontally: ${item.path}`);
      if (item.path.startsWith('/tutorials/') && layout.scroll_regions.length) throw new Error(`A tutorial example still requires horizontal scrolling: ${item.path}`);
      const images = await page.locator('article img').evaluateAll(items => items.map(item => ({ source: item.getAttribute('src'), complete: item.complete, width: item.naturalWidth, height: item.naturalHeight })));
      if (images.some(item => !item.complete || !item.width || !item.height)) throw new Error(`An image failed to load: ${item.path}`);
      if (!item.localReadme) {
        const links = await page.locator('article a[href]').evaluateAll(items => items.map(item => item.href));
        for (const link of links) {
          const url = new URL(link);
          if (url.origin === origin) { url.hash = ''; localLinks.add(url.href); }
        }
      }
      pages.push({ path: item.path, status: response.status(), layout, images, screenshot: item.image, rendering: item.localReadme ? 'Local Python Markdown preview, not GitHub rendering.' : 'MkDocs Material' });
    }
    const links = [];
    for (const url of localLinks) {
      const response = await page.request.get(url);
      links.push({ path: new URL(url).pathname, status: response.status() });
      if (response.status() !== 200) throw new Error(`A local documentation link failed: ${url}`);
    }
    const exampleResponse = await page.request.get(`${origin}/tutorials/examples/p06-view-apply.json`);
    const example = await exampleResponse.json();
    if (example.width !== 1200 || example.height !== 800 || example.context.property.name !== 'PRESSURE') throw new Error('The documented full request did not load correctly.');
    if (errors.length) throw new Error(`Browser page errors: ${errors.join('; ')}`);
    const record = {
      reviewed_at: new Date().toISOString(),
      source_commit: execFileSync('git', ['-C', repository, 'rev-parse', 'HEAD'], { encoding: 'utf8' }).trim(),
      working_tree: execFileSync('git', ['-C', repository, 'status', '--short'], { encoding: 'utf8' }),
      browser: browser.version(),
      playwright: require('playwright/package.json').version,
      node: process.version,
      pages,
      links,
      full_request_status: exampleResponse.status(),
      page_errors: errors,
      visual_review_required: true,
    };
    fs.writeFileSync(path.join(output, 'review.json'), JSON.stringify(record, null, 2) + '\n');
    console.log(JSON.stringify({ source_commit: record.source_commit, pages: pages.length, links: links.length, page_errors: errors, images: pages.flatMap(item => item.images) }, null, 2));
  } finally {
    if (browser) await browser.close();
    server.kill('SIGTERM');
    fs.writeFileSync(path.join(output, 'server.log'), serverLog);
  }
})().catch(error => { console.error(error); process.exitCode = 1; });
